========================================================
==    $(reset)$(c6)Controls                                       $(reset)$(dim) ==
========================================================$(reset)

Movement - Keyboard
- $(c6)wasd / arrows$(reset) -> Move player
- $(c6)space bar    $(reset) -> Jump
- $(c6)r            $(reset) -> Restart level (counts as a death)

Movement - Gamepad
- $(c6)d-pad        $(reset) -> Move player
- $(c6)south button $(reset) -> Jump / Confirm
- $(c6)start        $(reset) -> Confirm
- $(c6)west button  $(reset) -> Restart level (counts as a death)

Utility
- $(c6)num 1        $(reset) -> Toggle post-processing effects
- $(c6)num 2        $(reset) -> Save screenshot to current directory
- $(c6)num 3        $(reset) -> Randomize level color palette
- $(c6)num 4        $(reset) -> Toggle debug statistics
- $(c6)num 8        $(reset) -> Lower master volume by 5%
- $(c6)num 9        $(reset) -> Raise master volume by 5%
- $(c6)m            $(reset) -> Mute audio
- $(c6)- / _        $(reset) -> Shrink image by one column
- $(c6)+ / =        $(reset) -> Grow image by one column
